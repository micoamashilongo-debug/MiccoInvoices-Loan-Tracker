-- MiccoInvoices Loan Tracker: cloud-sync table -- Run this script in Supabase Dashboard > SQL Editor. -- It stores each seller's customer and loan arrays separately by authenticated user ID. create table if not exists public.seller_app_data ( user_id uuid primary key references auth.users(id) on delete cascade, customers jsonb not null default '[]'::jsonb, loans jsonb not null default '[]'::jsonb, updated_at timestamptz not null default now() ); alter table public.seller_app_data enable row level security; drop policy if exists "Sellers can read their own app data" on public.seller_app_data; create policy "Sellers can read their own app data" on public.seller_app_data for select to authenticated using (auth.uid() = user_id); drop policy if exists "Sellers can insert their own app data" on public.seller_app_data; create policy "Sellers can insert their own app data" on public.seller_app_data for insert to authenticated with check (auth.uid() = user_id); drop policy if exists "Sellers can update their own app data" on public.seller_app_data; create policy "Sellers can update their own app data" on public.seller_app_data for update to authenticated using (auth.uid() = user_id) with check (auth.uid() = user_id); drop policy if exists "Sellers can delete their own app data" on public.seller_app_data; create policy "Sellers can delete their own app data" on public.seller_app_data for delete to authenticated using (auth.uid() = user_id); grant select, insert, update, delete on public.seller_app_data to authenticated; -- Enable realtime updates so logged-in devices can receive changes. do $$ begin alter publication supabase_realtime add table public.seller_app_data; exception when duplicate_object then null; when undefined_object then raise notice 'Publication supabase_realtime was not found; enable Realtime for seller_app_data in the Supabase dashboard.'; end $$;
